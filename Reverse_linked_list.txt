Given the head of a singly linked list, reverse the list, and return the reversed list.

solution in python3 =>

class Solution:
    def reverseList(self, head: Optional[ListNode]) -> Optional[ListNode]:
        if head is None:
            return None
        l=head
        a=[]
        while(head!=None):
            a.append(head.val)
            head=head.next
        head=l
        for i in range(len(a)-1,-1,-1):
            head.val=a[i]
            head=head.next
        return l
