Given the head of a linked list, return the list after sorting it in ascending order.

solution in python3 =>

class Solution:
    def sortList(self, head: Optional[ListNode]) -> Optional[ListNode]:
        l=head
        a=[]
        while(head!=None):
            a.append(head.val)
            head=head.next
        a.sort()
        head=l
        i=0
        while(head!=None):
            head.val=a[i]
            i+=1
            head=head.next
        head=l
        return head
